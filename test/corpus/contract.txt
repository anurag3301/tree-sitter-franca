package demo

interface Door {
  method open { }
  method close { }
  attribute Boolean locked
  broadcast opened { }

  contract {
    vars {
      Int32 count = 0
      Boolean armed;
    }
    PSM {
      initial Closed
      state Closed {
        on call open [!locked] -> Open
        on set locked -> Closed
      }
      state Open {
        on call close -> Closed {
          count = count + 1
        }
        on respond open:other -> Open
        on signal opened -> Open
        on update locked -> Open
        on error open -> Closed
      }
    }
  }
}
